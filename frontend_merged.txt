export function snakeToCamel(str) {
    return str.replace(/([-_][a-z0-9])/gi, (group) =>
    group.toUpperCase().replace('-', '').replace('_', ''));
}

export function camelToSnake(str) {
    if (typeof str !== 'string') return str;
    return str.replace(/([A-Z])/g, ($1) => `_${$1.toLowerCase()}`);
}

function isPlainObject(obj){
    return (
        obj !==null &&
        typeof obj === 'object' &&
        Object.prototype.toString.call(obj) === '[object Object]'
    )
}

export function adaptToCamel(data) {
    if (!data) return data;

    if (Array.isArray(data)){
        return data.map((item) => adaptToCamel(item));
    }

    if(isPlainObject(data)) {
        const result = {};
        for (const [key, value] of Object.entries(data)){
            result[snakeToCamel(key)] = adaptToCamel(value);
        }
        return result;
    }

    return data;
}

export function adaptToSnake(data) {
    if (!data) return data;

    if (Array.isArray(data)) {
        return data.map((item) => adaptToSnake(item));
    }

    if (isPlainObject(data)) {
        const result = {};
        for (const [key, value] of Object.entries(data)) {
        result[camelToSnake(key)] = adaptToSnake(value);
        }
        return result;
    }

    return data;
}

// src/api/client.js
import axios from 'axios';
import {  getAccess, getRefresh, setAccess, clearAll,} from '../utils/storage';
import { ENDPOINTS, REQ } from '../config/apiContract';


import { demoAdapter } from './demoAdapter';
import { adaptToSnake, adaptToCamel } from './adapter';

const DEMO_MODE = import.meta.env.VITE_DEMO_MODE === 'true';  // DEMO

const apiClient = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 10000,
  headers: { 'Content-Type': 'application/json' },
  ...(DEMO_MODE ? { adapter: demoAdapter } : {}),  //DEMO
});

// Request interceptor 
apiClient.interceptors.request.use((config) => {
  const access = getAccess();
  if (access && !config.headers.Authorization) {         // 08-01  automount JWT Bearer Token
    config.headers.Authorization = `Bearer ${access}`;
  }

  if (config.data && !(config.data instanceof FormData) && !config.skipTransform) {

    config.data = adaptToSnake(config.data)
  }

  if ( config.params && typeof config.params === 'object' && !config.skipTransform) {
    config.params = adaptToSnake(config.params);
    
  }
  //config.headers['X-API-Version'] = API_VERSION;
  return config;
}, 
(error)=> Promise.reject(error));

// Singleton refresh promise 
let refreshPromise = null;

const doRefresh = async () => {
  const refresh = getRefresh();
  if (!refresh) throw new Error('No refresh token');
  const { data } = await axios.post(
    `${import.meta.env.VITE_API_BASE_URL || 'http://127.0.0.1:8000'}${ENDPOINTS.refresh}`,
    { [REQ.refresh]: refresh }
  );

  
  setAccess(data.access);
  return data.access;
};

// Endpoints that should NOT trigger refresh
const AUTH_ENDPOINTS = [
  ENDPOINTS.login,
  ENDPOINTS.register,
  ENDPOINTS.login2FA,
  ENDPOINTS.refresh,
];

apiClient.interceptors.response.use(
  /*(res) => res.data,      change 08-01*/
(response) => {
    // 204 No Content retrun null
    if (response.status === 204) {
      return null;
    }

    const payload = response.data;
    if (response.config?.skipTransform) {
      return payload;
    }

    // auto change to Camel
    return adaptToCamel(payload);
  },

  async (error) => {
    const original = error.config;
    const status = error.response?.status;

    // ★ Skip refresh logic for auth endpoints 
    const isAuthEndpoint = AUTH_ENDPOINTS.some((ep) =>
      original?.url?.includes(ep)
    );

    if (status === 401 && !original._retry && !isAuthEndpoint) {
      original._retry = true;
      try {
        if (!refreshPromise) {
          refreshPromise = doRefresh().finally(() => { refreshPromise = null; });
        }
        const newAccess = await refreshPromise;
        original.headers.Authorization = `Bearer ${newAccess}`;
        return apiClient(original);
      } catch {
        clearAll();
        if (!window.location.pathname.startsWith('/login')) {
            window.location.href = '/login';
        }
          return Promise.reject({ message: 'Session expired, please login again' });
        }
    }

    // ★ Extract error message 
    const data = error.response?.data;
    let message = 'Network error';
      if (data) {     /* 08-01 */
        if (typeof data === 'string') {
          message = data;
        } else if (data.message) {
          message = data.message;
        } else if (data.detail) {
          message = data.detail;
        } else if (typeof data === 'object') {
          
          const firstKey = Object.keys(data)[0];
          if (firstKey) {
            const fieldError = data[firstKey];
            message = Array.isArray(fieldError)
              ? `${firstKey}: ${fieldError[0]}`
              : `${firstKey}: ${fieldError}`;
          }
        }
      } else if (error.message) {
        message = error.message;
      }

    return Promise.reject({ message, status, raw: error });
  }
);

export default apiClient;

const buildResponse = (data, status = 200) => ({
  data,
  status,
  statusText: 'OK',
  headers: {},
  config: {},
});

const buildError = (message, status = 400) => {
  const error = new Error(message);
  error.response = {
    data: { message },
    status,
    statusText: 'Error',
    headers: {},
    config: {},
  };
  return Promise.reject(error);

/**
 * Route handlers / 路由处理
 */
const routeHandlers = [
  // ============ Login ============
  // 演示：识别 username 里的关键字决定角色
  {
    match: (m, u) => m === 'post' && u.endsWith('/accounts/login/'),
    handle: async (config) => {
      await delay();
      const body = typeof config.data === 'string' ? JSON.parse(config.data) : config.data;
      const username = body?.username || '';

      let user = DEMO_USERS.user;
      if (username.includes('owner')) user = DEMO_USERS.owner;
      else if (username.includes('admin')) user = DEMO_USERS.admin;

        access: MOCK_ACCESS,
        refresh: MOCK_REFRESH,

  // ============ Refresh ============
  {
    match: (m, u) => m === 'post' && u.endsWith('/api/token/refresh/'),
    handle: async () => {
      await delay(100);
      return buildResponse({ access: MOCK_ACCESS });
    },
  },

  // ============ Modules ============
  {
    match: (m, u) => m === 'get' && u.endsWith('/modules/'),

      await delay();

export const demoAdapter = async (config) => {
  const method = (config.method || 'get').toLowerCase();
  const url = config.url || '';

  for (const route of routeHandlers) {
    if (route.match(method, url)) {
      try {
        return await route.handle(config);
      } catch (err) {
        return buildError(err.message || 'Mock error', 500);
      }
    }
  }

  return buildError(`Demo mode: no mock for ${method.toUpperCase()} ${url}`, 404);
};

function CodeInput({onSubmit, submitLabel ='Verity', loadingLabel = 'Verifying', autoFocus = true,}){

    const [code,setCode] =  useState('');
    const [error,setError] = useState('');
    const [isLoading, setIsLoading] = useState(false);

    const inputRef = useRef(null);


    // auto-focus on mount
    useEffect ( () => { if (autoFocus) inputRef.current?.focus(); },[autoFocus]); /* if inputRef.current is not null/undefined， call on focus. if is return undefined. */

    const handleChange = (e) =>{ setCode(filterCode(e.target.value));};
    const handleSubmit = async (e)=> {
        e.preventDefault();
        setError('');

        const formatErr = validateCode(code);
        if(formatErr) {
            setError(formatErr);
            return;
        }

        setIsLoading(true);
        try{
            await onSubmit(code);
        } catch (err) {
            setError(err.message || 'Verification failed');
            setCode('');
            inputRef.current?.focus();

        } finally { setIsLoading(false);}
    };

    return(
    
        <form onSubmit={handleSubmit}>
            <Stack spacing={2.5}>
                {error && <AlertMessage type="error">{error}</AlertMessage>}

                <FormField
                label="Verification Code"
                value={code}
                onChange={handleChange}
                
                disabled={isLoading}
                inputRef={inputRef}

                    inputMode: 'numeric',

                loading={isLoading}
                loadingText={loadingLabel}

import { useState } from 'react';

import EmptyState from '../ui/EmptyState';

/**
 * Section manager - add/edit/delete sections inline  -> 新增/编辑/删除
 *
 * @param {Object} props
 * @param {number|string} props.moduleId
 * @param {Array} props.initialSections
 */
function SectionManager({ moduleId, initialSections = [] }) {

    const [sections, setSections] = useState(initialSections);
    const [editingId, setEditingId] = useState(null);
     const [error, setError] = useState('');
     const [isSaving, setIsSaving] = useState(false);    /* 08-01 */

    const [draft, setDraft] = useState({ title: '', body: '', order: 1 });
   

    const isCreating = editingId === 'new';     // true， if creating a new section


    const startCreate = () => {         // Start creating a new section
        setEditingId('new');
        setDraft({ title: '', body: '',  order: sections.length + 1 });
    };

    const startEdit = (section) => {    //Start editing an existing section
        setEditingId(section.id);
        setDraft({title: section.title,  body:section.body, order:section.order});
    };

    const cancel = () => {              //Cancel creating & editing

        setEditingId(null);
        setDraft({ title: '', body: '', order: 1 });
        setError('');
    };


    const handleSave = async () => {    // Save new or existing section 
        
        if (!draft.title.trim()) {

        setError('Title is required');
        return;
        }

        setIsSaving(true);   /* 08-01 */
        setError('');

        try {
        if (isCreating) {

            // future: const created = await sectionsApi.createSection({ module: moduleId, ...draft });
            /*const created = { id: Date.now(), module: moduleId, ...draft };
            await new Promise((r) => setTimeout(r, 300));
            */
            const created = await sectionsApi.createSection({
                    module: moduleId,
                    title: draft.title,
                    body: draft.body,
                    order: draft.order,
                });
            setSections((prev) => [...prev, created]);
        } else {

            // future: await sectionsApi.updateSection(editingId, draft);
            /*await new Promise((r) => setTimeout(r, 300)); */
            const updated = await sectionsApi.updateSection(editingId, {
                    title: draft.title,
                    body: draft.body,
                    order: draft.order,
                });
            setSections((prev) =>           // Update the section in the list
            prev.map((s) => (s.id === editingId ? updated : s))      //https://react.dev/learn/updating-arrays-in-state
            );
        }
        cancel();
        } catch (err) {
        setError(err.message || 'Failed to save section');
        } finally {
            setIsSaving(false);
        }
    };

    /*  Delete section   */
    const handleDelete = async (sectionId) => {

        if (!window.confirm('Delete this section?')) return;
        
        const prev = sections;
        setSections((items) => items.filter((s) => s.id !== sectionId));

        try {

        // future: await sectionsApi.deleteSection(sectionId);
        /*await new Promise((r) => setTimeout(r, 300));*/
        await sectionsApi.deleteSection(sectionId);

        } catch (err) {
            setSections(prev);
            setError(err.message || 'Failed to delete');
        }
    };

    return (
        <Stack spacing={2.5}>
        {error && <Alert severity="error">{error}</Alert>}

        <Box sx={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
            <Typography variant="h3"> Sections ({sections.length}) </Typography>

            {!isCreating && (
                <ActionButton fullWidth={false} variant="secondary" onClick={startCreate}>
                    <Box sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
                    <Icon.Add fontSize="small" /> Add Section  </Box>
                </ActionButton>
            )}
        </Box>


        {isCreating && (    // creating  form
            <Paper elevation={0} sx={{ p: 2, border: '2px solid', borderColor: 'primary.main' }}>
                <Stack spacing={2}>
                    <FormField  label="Title"   value={draft.title} onChange={(e) => setDraft((d) => ({ ...d,  title:e.target.value }))}  disabled={isSaving} />
                    <FormField  label="Body"    value={draft.body}  onChange={( e) => setDraft((d) => ({ ...d, body: e.target.value }))}                            inputProps={{ multiline: true, rows: 4 }} disabled={isSaving}   />
                    <FormField  label="Order"   type="number"   value={draft.order} onChange={(e) => setDraft((d) => ({ ...d, order: Number(e.target.value) }))}    inputProps={{ min: 1 }}  disabled={isSaving} />

import { useState } from 'react';
import { Stack, Paper, Typography, Box, Chip, Grid } from '@mui/material';
import LinkIcon from '@mui/icons-material/Link';

import { Alert } from '@mui/material';

/**
 * Matching question - Check future: https://mui.com/material-ui/react-grid/  react-stack; react-paper; react-chip for layout
 *https://mui.com/material-ui/react-transfer-list/
 *https://mui.com/material-ui/react-paper/#variants
 *https://mui.com/material-ui/react-list/#simple-list

 * Interaction:
 *   1. Click a left item → selected (highlighted)
 *   2. Click a right item → establishes pair
 *   3. Click an already-paired item → unpair
 *   -----------
 *   1. 点击左侧项 → 选中
 *   2. 点击右侧项 → 建立配对
 *   3. 点击已配对项 → 取消配对
 *
 * Data shape  
 *   question.pairs = [ { id: 31, left: 'Phishing', right: 'Email attack' } ]
 *   value = { [leftPairId]: rightPairId }  // e.g. { 31: 32, 32: 31 }
 *
 * @param {Object} props
 * @param {object} props.question
 * @param {Object} props.value
 * @param {(v: object) => void} props.onChange
 * @param {boolean} [props.disabled]
 */
function MatchingQuestion({ question, value = {}, onChange, disabled = false }) {
    const pairs = question.pairs || [];

    const [rightItems] = useState(() =>   // Shuffle right column (stable per mount)- check 
        [...pairs].sort( () => Math.random()-0.5 )
    );

    const [selectedLeft, setSelectedLeft] = useState(null);

    const handleLeftClick = (pairId) => {
        if (disabled) return;


            if (value[pairId] != null)  { // This left item is already paired → unpair it
                const next = { ...value  };
                delete next[pairId];
                onChange(next);
                setSelectedLeft(null);
                return;
            }

        setSelectedLeft( pairId === selectedLeft? null : pairId);
    };

    const handleRightClick = (pairId) => {
        if (disabled) return;


            const existingLeft = Object.keys(value ).find((k) => value[k]=== pairId);             // This right item is already taken → free it
            if (existingLeft) {
                const next = {  ...value};

                delete next[existingLeft];

                onChange(next);
                return;
            }

        if (selectedLeft == null) return;    // No left selected → ignore

       
        const next = { ...value, [selectedLeft]: pairId };       // Establish pair

        onChange(next);
        setSelectedLeft(null);
    };

    return (
        <Box>
            <Alert severity="info" sx={{ mb: 2 }}>
                This section is only a front-end demonstration and has not yet been integrated with the back-end marking API. Your connection result will not be saved.
            </Alert>

        <Typography variant="body2" color="text.secondary" sx={{ mb: 2 }}>
            Tap a term on the left, then tap its match on the right.
        </Typography>

        <Grid container spacing={2}>

            {/* ******* Left */}

            <Grid item xs={6}>
            <Typography variant="caption" sx={{ mb:1, display:  'block', fontWeight: 600 }}>
                TERMS
            </Typography>
            <Stack spacing={1}>
                {pairs.map((pair) => {
                const isPaired = value[pair.id] != null;
                const isSelected = selectedLeft === pair.id;

                return (
                    <Paper
                    key={pair.id}
                    elevation={0}
                    onClick={() => handleLeftClick(pair.id)}
                    sx={{
                        p: 1.5,
                        border: '2px solid',
                        borderColor: isSelected ?'primary.main':isPaired ? 'primary.light' : 'divider',
                        backgroundColor: isSelected ? 'rgba(59, 91, 140,  0.06 )' : isPaired ? 'rgba( 59,  91, 140, 0.03)' : '#fff',
                        cursor: disabled ? 'default' : 'pointer',
                        transition: 'all 0.15s',
                        '&:hover': disabled ? {} : { borderColor: 'primary.light' },
                    }}
                    >
                    <Typography variant="body1">{pair.left}</Typography>
                    </Paper>
                );
                })}
            </Stack>
            </Grid>

            {/* ******* Right */}

            
            <Grid item xs={6}>
            <Typography variant="caption" sx={{ mb: 1, display: 'block', fontWeight: 600 }}>
                DESCRIPTIONS
            </Typography>
            <Stack spacing={1}>
                {rightItems.map((pair) => {
                const pairedToLeft = Object.keys(value).find((k) => value[k]===pair.id);
                const isPaired = !!pairedToLeft;

                return (
                    <Paper
                    key={pair.id}
                    elevation={0}
                    onClick={() => handleRightClick(pair.id)}
                    sx={{
                        p: 1.5,
                        border: '2px solid',
                        borderColor: isPaired ? 'primary.light' : 'divider',
                        backgroundColor: isPaired ? 'rgba(59, 91, 140, 0.03)' : '#fff',
                        cursor: disabled ? 'default' : 'pointer',
                        transition: 'all 0.15s',
                        '&:hover': disabled ? {} : { borderColor: 'primary.light' },
                    }}
                    >
                    <Typography variant="body2">{pair.right}</Typography>
                    </Paper>
                );
                })}
            </Stack>
            </Grid>
        </Grid>

        {Object.keys(value).length > 0 && (
            <Box sx={{ mt: 3 }}>
                <Typography variant="caption" color="text.secondary" sx={{ mb: 1, display: 'block' }}> Matched ({Object.keys(value).length}):  </Typography>
            <Stack direction="row" spacing={1 } sx={{ flexWrap:'wrap', gap: 1 }}>
                {Object.entries(value).map(([leftId, rightId]) => {

                    const left = pairs.find((p) => p.id === Number(leftId));
                    const right = pairs.find((p) => p.id === rightId);
                    return (
                        <Chip
                            key={leftId}
                            icon={<LinkIcon fontSize="small" />}
                            label={`${left?.left} → ${right?.right}`}
                            size="small"
                            onDelete={  disabled ? undefined : () => { const next = { ...value };
                                delete next[leftId];
                                onChange(next);
                                }
                            }
                        />
                    );

                })}
            </Stack>
            </Box>
        )}


        </Box>
    );
}


export default MatchingQuestion;

 * @param {(v: number[]) => void} props.onChange

 */
function MultipleChoiceQuestion({ question, value = [], onChange, disabled = false }) {
    const sorted = [...(question.choices || [])].sort((a, b) => a.order - b.order);

    const toggle = (choiceId) => {
        if (disabled) return;

        const next = value.includes(choiceId) ? value.filter((id) => id !== choiceId) : [...value, choiceId]; // toggle selection
        onChange(next);
    };

    return (
        <Stack spacing={1}>

            {sorted.map((choice) => {
                const isSelected = value.includes(choice.id);

                return (

                    <Paper
                        key={ choice.id}
                        elevation={0}
                        onClick={() =>toggle(choice.id ) }
                        sx={{
                        p: 1.5,
                        display: 'flex',

                        borderColor: isSelected ? 'primary.main' : 'divider',

import { Button, CircularProgress } from "@mui/material";


/**
 * Action button with loading state
 *
 * @param {Object} props
 * @param {React.ReactNode} props.children
 * @param {'primary' | 'secondary' | 'danger' | 'ghost'} [props.variant='primary']
 * @param {boolean} [props.loading=false]
 * @param {string} [props.loadingText]
 * @param {boolean} [props.disabled]
 * @param {() => void} [props.onClick]
 * @param {'button' | 'submit' | 'reset'} [props.type='button']
 * @param {boolean} [props.fullWidth=true]
 * @param {Object} [props.sx]
 */


function ActionButton ({
    children,
    variant='primary',
    loading = false,
    loadingText,
    disabled,
    onClick,
    type='button',
    fullWidth =true,
    sx,
})
{
    const variantMap = {
    primary: 'contained',
    secondary: 'outlined',
    danger: 'contained',
    ghost: 'text',
    };

    const muiVariant = variantMap[variant] || 'contained';

    const color = variant === 'danger' ? 'error' : 'primary';

    return(
        <Button type={type}
        variant={muiVariant}
        color={color}
        onClick={onClick}
        disabled={disabled || loading}
        fullWidth={fullWidth}
        size="large"
        startIcon={loading ? <CircularProgress size={18} color="inherit" /> : null}
        sx={sx}>

        {loading && loadingText ? loadingText : children}
        </Button>        
    )
}

export default ActionButton;

import { useState, useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import { Stack, Box, Typography, Paper,  } from '@mui/material';
import * as authApi from '../../api/auth';
import { useAuth } from '../../hooks/useAuth';
import { PageContainer, PageTitle, ActionButton, AlertMessage, Icon } from '../../components/ui';
import CodeInput from '../../components/auth/CodeInput';
import './Setup2FAPage.css';


/*
 PyOTP handbook : https://pyauth.github.io/pyotp/
 */
function Setup2FAPage() {
  const navigate = useNavigate();
  const { refreshUser } = useAuth();
  // Setup data 
  const [secret,setSecret]=useState('');
  const [qrCode,setQrCode]=useState('');

  const [setupError,setSetupError] = useState('');
  const [isLoadingSetup,setIsLoadingSetup] = useState(true);

  // Success state 
  const [isEnabled,setIsEnabled] = useState(false);

  // generate secret + fetch QR
  useEffect(() => {
    let ignore = false; 

    const load = async () => {
      try {
        // Step 1: generate secret 
       // await authApi.setup2FA();
        const setupRes = await authApi.setup2FA();
        

        // Step 2: get QR
       
        const qrRes = await authApi.fetch2FAQR();
        if (!ignore) {
          setSecret(setupRes.secret);
          setQrCode(qrRes.qrCode);
        }

        // Also fetch secret for manual entry 
        //const setupRes = await authApi.setup2FA();   /* put at here, setup is called 2 times */
        //setSecret(setupRes.secret);
      } catch (err) {
        if (!ignore) { setSetupError(err.message || 'Failed to load 2FA setup');  }
      } finally {
        if (!ignore) {setIsLoadingSetup(false);}
      }
    };
    load();
    return () => { ignore = true; }; 
  }, []);

  /**
   * Submit OTP to verify and enable 
   * @param {string} otp
   */
  const handleSubmit = async (otp) => {
    await authApi.verify2FA(otp);
    await refreshUser();
    setIsEnabled(true);
  };



  if (isLoadingSetup) {
    return (
      <PageContainer className="setup-2fa-page" maxWidth="sm">
        <Paper elevation={0} sx={{ p: { xs: 2.5, sm: 3, md: 4 }, borderRadius: 2, textAlign: 'center' }}>
          <Typography>Loading 2FA setup...</Typography>
        </Paper>
      </PageContainer>
    );
  }


  if (setupError) {
    return (
      <PageContainer className="setup-2fa-page" maxWidth="sm">
        <Paper elevation={0} sx={{ p: { xs: 2.5, sm: 3, md: 4 }, borderRadius: 2 }}>
          <Stack spacing={2.5}>
            <PageTitle title="Setup Failed" />
            <AlertMessage type="error">{setupError}</AlertMessage>
            <ActionButton onClick={() => navigate('/account')}>
              Back to Account
            </ActionButton>
          </Stack>
        </Paper>
      </PageContainer>
    );
  }

  // Success
  if (isEnabled) {
    return (
      <PageContainer className="setup-2fa-page" maxWidth="sm">
        <Paper elevation={0} sx={{ p: { xs: 2.5, sm: 3, md: 4 }, borderRadius: 2 }}>
          <Stack spacing={2.5} alignItems="center">
            <Icon.CheckCircle sx={{ fontSize: 64, color: 'success.main' }} />
            <PageTitle
              title="2FA Enabled"
              subtitle="Two-factor authentication is now active for your account."
            />
            <ActionButton onClick={() => navigate('/account')}>
              Back to Account
            </ActionButton>
          </Stack>
        </Paper>
      </PageContainer>
    );
  }


  return (
    <PageContainer className="setup-2fa-page" maxWidth="sm">
      <Paper elevation={0} sx={{ p: { xs: 2.5, sm: 3, md: 4 }, borderRadius: 2 }}>
        <Stack spacing={3}>
          <PageTitle
            title="Set Up 2FA"
            subtitle="Scan the QR code with your Okta Verify app."
          />

          {/* QR code */}
          {qrCode && (
            <Box sx={{ textAlign: 'center' }}>
              <img
                src={`data:image/png;base64,${qrCode}`}
                alt="2FA QR Code"
                style={{
                  width: 200,
                  height: 200,
                  border: '1px solid #e0e0e0',
                  borderRadius: 8,
                }}
              />
            </Box>
          )}

          {/* Manual secret  */}
          <Box>
            <Typography variant="body2" color="text.secondary" gutterBottom>
              Can't scan? Enter this key manually:
            </Typography>
            <Box
              component="code"
              sx={{
                display: 'block',
                p: 1.5,
                backgroundColor: '#f5f5f7',
                borderRadius: 1,
                fontFamily: 'monospace',
                fontSize: '0.85rem',
                wordBreak: 'break-all',
              }}
            >
              {secret || 'Loading...'}
            </Box>
          </Box>

          {/* OTP input */}
          <CodeInput
            onSubmit={handleSubmit}
            submitLabel="Verify & Enable"
            loadingLabel="Enabling..."
          />

          <ActionButton variant="ghost" onClick={() => navigate('/account')}>
            Cancel
          </ActionButton>
        </Stack>
      </Paper>
    </PageContainer>
  );
}

export default Setup2FAPage;

import './RegisterPage.css';





const REGISTER_FIELDS = [
    { name: 'username',        label: 'Username',         type: 'text',     placeholder: 'Enter your username' },
    { name: 'email',           label: 'Email',            type: 'text',     placeholder: 'Enter your email' },
    { name: 'password',        label: 'Password',         type: 'password', placeholder: 'Enter your password' },
    { name: 'confirmPassword', label: 'Confirm Password', type: 'password', placeholder: 'Re-enter your password' },
]; /* rewrite 5-01, change partial assignment name:username laber:username  */

const EMAIL_RULES = [
    { test: (v) => !!v.trim(), msg: 'Email is required' },
    { test: (v) => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v), msg: 'Email format is invalid' },
];

function RegisterPage() {
    const navigate = useNavigate();


    const [form, setForm] = useState({
    username: '',
    email: '',
    password: '',
    confirmPassword: '',
    });

    /* same to LoginPage, and keep they at here */
    const [errors, setErrors] = useState({});
    const [isLoading, setIsLoading] = useState(false);   
    const [serverError, setServerError] = useState('');

    const updateField = (name,value) => { 
        setForm((prev) => ({...prev, [name]:value}));
     }

     const validate = ( ) => { 
        const newError = {};

        if(!form.username.trim()) {
            newError.username = 'Username is required';   //* debuge .name issue,   6-03 */
        }

        for(const rule of EMAIL_RULES) {
            if(!rule.test(form.email)) {
                newError.email=rule.msg;
                break;
            }
        }

        const pwdErr = validatePassword(form.password)      // shared rules with -passwordrules
        if(pwdErr) newError.password = pwdErr;

        //confirm password
        if(!form.confirmPassword) {
            newError.confirmPassword = 'Please confirm your password!';

        } else if (form.confirmPassword !== form.password) {
            newError.confirmPassword = 'Password incorrect';
        }

        return newError;
    };

    const handleSubmit = async (ee) => { 
        ee.preventDefault();
        setServerError('');

        const newErr = validate();
        if(Object.keys(newErr).length > 0) {
            setErrors(newErr);
            return;
        }
        setErrors({});

        setIsLoading(true);
        try{ /* rewrite 5-01, try */
            
            await authApi.register({
                username: form.username,
                email: form.email,
                password: form.password
            });
            navigate('/login', {replace:true});  
        } catch (err) {
            setServerError(err.message || 'Registration failed!');
        } finally {
            setIsLoading(false);
        }
    };
 
    return(
    <AuthLayout className="register-page" title="Register" subtitle="Create your account.">     {/* debug  title="Regiser"*/}
        <form onSubmit={handleSubmit}>
            <Stack spacing={2.5}>
                {serverError && <AlertMessage type="error">{serverError}</AlertMessage>}

                {REGISTER_FIELDS.map((field) => (
                    <Box key={field.name}>
                        <FormField 
                        label={field.label}
                        type={field.type}
                        value={form[field.name]}
                        onChange={(e) => updateField(field.name, e.target.value)}
                        error={errors[field.name]}
                        disabled={isLoading}
                        />

                        {field.name ==="password" && form.password.length > 0 && (
                            <List dense sx={{mt:1,pl:1}}>
                                {getActiveRules().map((rule) => {
                                const passed = rule.test(form.password);
                                return (
                                <ListItem key={rule.key} disableGutters sx={{ py: 0 }}>
                                    <ListItemIcon sx={{ minWidth: 28 }}>
                                    {passed
                                        ? <Icon.CheckCircle fontSize="small" color="success" />
                                        : <Icon.RadioButtonUnchecked fontSize="small" sx={{ color: 'text.disabled' }} />}
                                    </ListItemIcon>
                                    <ListItemText
                                    primary={rule.label}
                                    primaryTypographyProps={{
                                        variant: 'body2',
                                        color: passed ? 'success.main' : 'text.secondary',
                                    }}
                                    />
                                </ListItem>
                                );
                            })}
                            </List>
                        )}
                    </Box>
                    ))}

                    <ActionButton type="submit" loading={isLoading} loadingText="Registering...">
                        Register
                    </ActionButton>

                    <MuiLink
                        component={RouterLink}
                        to="/login"
                        underline="hover"
                        sx={{ textAlign: 'center', fontSize: '0.9rem' }}
                    >
                        Already have an account? Login
                    </MuiLink>
                    </Stack>
        </form>
    </AuthLayout>
  );
}

export default RegisterPage;

  Box,
  Paper,

  CircularProgress,
  Alert,
} from "@mui/material";

import SchoolIcon from "@mui/icons-material/School";
import ArrowForwardIcon from "@mui/icons-material/ArrowForward";

import { PageContainer } from "../../components/ui";
import { getModules, getEnrollments } from "../../api/modules";

import "./ModulesPage.css";

function ModulesPage() {
  const location = useLocation();
 
  const isMyModules = location.pathname.startsWith("/user/modules");

  const basePath = isMyModules ? "/user/modules" : "/modules";

  const [modules, setModules] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");


  
  useEffect(() => {
    let active = true;

    const loadModules = async () => {
      setLoading(true);
      setError("");

      try {
        const allModules = await getModules();

        let visibleModules = allModules;


        if (isMyModules) {
          const enrollments = await getEnrollments();

          const enrolledIds = new Set(
            enrollments.map((enrollment) =>
              Number(enrollment.module)
            )
          );

          visibleModules = allModules.filter((module) =>
            enrolledIds.has(Number(module.id))
          );
        }

        if (active) {
          setModules(visibleModules);
        }
      } catch (err) {
        if (active) {
          setError(err.message || "Failed to load modules");
        }
      } finally {
        if (active) {
          setLoading(false);
        }
      }
    };

    loadModules();

    return () => {
      active = false;
    };
  }, [isMyModules]);

  return (

            {isMyModules ? "My Modules" : "Browse Modules"}

              : "Explore available cybersecurity training modules."}

          <Box
            sx={{
              display: "flex",
              justifyContent: "center",
              py: 5,
            }}
          >
            <CircularProgress />

        )}

        {!loading && error && (<Alert severity="error">
            {error}

import { useState, useEffect } from 'react';

import * as modulesApi from '../../../api/modules';
import apiClient from '../../../api/client';
import { PageContainer, ActionButton, Icon, FormField } from '../../../components/ui';

const generateId = () => Math.random().toString(36).substring(2, 10);

function ManageQuizPage() {
  const { id: moduleId } = useParams();
  const navigate = useNavigate();

  const [module, setModule] = useState(null);
  const [questions, setQuestions] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [isSaving, setIsSaving] = useState(false);
  const [error, setError] = useState('');
  const [success, setSuccess] = useState('');

  useEffect(() => {
    let ignore = false;
    const load = async () => {
      setIsLoading(true);
      try {
        const moduleData = await modulesApi.getModule(moduleId);
        
        let existingQuestions = [];
        try {
          const res = await apiClient.get(`/modules/${moduleId}/questions/`);
          existingQuestions = res || [];
        } catch {
          console.log('There is no question data yet for this module, starting fresh.');
        }

        if (!ignore) {
          setModule(moduleData);
          const formattedQs = existingQuestions.map((q) => ({
            ...q,
            choices: q.choices || [],
          }));
          setQuestions(formattedQs);
        }
      } catch (err) {
        if (!ignore) setError(err.message || 'Failed to load module or questions');
      } finally {
        if (!ignore) setIsLoading(false);
      }
    };
    load();
    return () => {
      ignore = true;
    };
  }, [moduleId]);


  const handleAddQuestion = () => {
    const newQ = {
      id: generateId(),
      text: '',
      question_type: 'SINGLE',
      points: 1,
      order: questions.length + 1,
      explanation: '',
      choices: [
        { id: generateId(), text: '', is_correct: true, order: 1 },
        { id: generateId(), text: '', is_correct: false, order: 2 },
      ],
    };
    setQuestions([...questions, newQ]);
  };


  const handleRemoveQuestion = async (qIndex) => {
    if (!window.confirm('Are you sure you want to delete this question?')) return;

    const targetQ = questions[qIndex];

    if (targetQ && typeof targetQ.id === 'number') {
      try {
        await apiClient.delete(`/questions/${targetQ.id}/`);
      } catch (err) {
        setError(err.message || 'Failed to delete question from server');
        return;
      }
    }

    const newQs = [...questions];
    newQs.splice(qIndex, 1);
    setQuestions(newQs);
  };


  const updateQuestion = (qIndex, field, value) => {
    const newQs = [...questions];
    newQs[qIndex] = { ...newQs[qIndex], [field]: value };
    setQuestions(newQs);
  };


  const handleAddChoice = (qIndex) => {
    const newQs = [...questions];
    newQs[qIndex].choices.push({
      id: generateId(),
      text: '',
      is_correct: false,
      order: newQs[qIndex].choices.length + 1,
    });
    setQuestions(newQs);
  };


  const handleRemoveChoice = async (qIndex, cIndex) => {
    const targetChoice = questions[qIndex]?.choices?.[cIndex];

    if (targetChoice && typeof targetChoice.id === 'number') {
      try {
        await apiClient.delete(`/choices/${targetChoice.id}/`);
      } catch (err) {
        setError(err.message || 'Failed to delete option from server');
        return;
      }
    }

    const newQs = [...questions];
    newQs[qIndex].choices.splice(cIndex, 1);
    setQuestions(newQs);
  };


  const updateChoice = (qIndex, cIndex, field, value) => {
    const newQs = [...questions];
    const isSingle = newQs[qIndex].question_type === 'SINGLE';

    const newChoices = newQs[qIndex].choices.map((choice, index) => {
      if (field === 'is_correct' && isSingle) {
        return { ...choice, is_correct: index === cIndex };
      }
      if (index === cIndex) {
        return { ...choice, [field]: value };
      }
      return choice;
    });

    newQs[qIndex].choices = newChoices;
    setQuestions(newQs);
  };

  const handleSaveAll = async () => {
    setError('');
    setSuccess('');
    setIsSaving(true);

    try {

      for (let i = 0; i < questions.length; i++) {
        const q = questions[i];
        if (!q.text.trim()) throw new Error(`Question ${i + 1} is missing question text.`);
        if (!q.choices || q.choices.length === 0) throw new Error(`Question ${i + 1} must contain options.`);
        if (!q.choices.some((c) => c.is_correct)) throw new Error(`Question ${i + 1} does not have a correct answer selected.`);
        if (q.choices.some((c) => !c.text.trim())) throw new Error(`Question ${i + 1} has an empty option description.`);
      }


      const existingRemoteQuestions = await apiClient.get(`/modules/${moduleId}/questions/`);
      if (Array.isArray(existingRemoteQuestions)) {
        for (const eq of existingRemoteQuestions) {
          await apiClient.delete(`/questions/${eq.id}/`);
        }
      }


      for (let i = 0; i < questions.length; i++) {
        const q = questions[i];
        const qPayload = {
          module: Number(moduleId),
          text: q.text,
          question_type: q.question_type,
          explanation: q.explanation || '',
          points: Number(q.points) || 1,
          order: i + 1,
          is_active: true, 
        };

        const createdQ = await apiClient.post('/questions/', qPayload);
        const questionId = createdQ?.id;

        if (!questionId) {
          throw new Error('Failed to obtain Question ID from server.');
        }


        for (let j = 0; j < q.choices.length; j++) {
          const c = q.choices[j];
          await apiClient.post('/choices/', {
            question: questionId,
            text: c.text,
            is_correct: Boolean(c.is_correct),
            order: j + 1,
          });
        }
      }

      setSuccess('The quiz questions have been saved successfully!');
      setTimeout(() => navigate(`/owner/modules/${moduleId}/edit`), 1200);
    } catch (err) {
      const errorMsg = err.message || err.detail || 'Failed to save the quiz.';
      setError(errorMsg);
    } finally {
      setIsSaving(false);
    }
  };

  if (isLoading) {
    return (
      <PageContainer maxWidth="md">
        <Typography>Loading...</Typography>
      </PageContainer>
    );
  }

  return (
    <PageContainer maxWidth="md">
      <Stack spacing={3}>

        <Box sx={{ mb: 1 }}>
          <Button
            component={RouterLink}
            to={`/owner/modules/${moduleId}/edit`}
            startIcon={<Icon.ArrowBack fontSize="small" />}
            sx={{
              color: 'text.secondary',
              fontSize: '0.875rem',

              mb: 1.5,

        {error && <Alert severity="error">{error}</Alert>}
        {success && <Alert severity="success">{success}</Alert>}


        <Box sx={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
          <Typography variant="h3">Questions ({questions.length})</Typography>

            <Box sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
              <Icon.Add fontSize="small" /> Add Question
            </Box>
          </ActionButton>
        </Box>

        <Stack spacing={3}>
          {questions.map((q, qIndex) => (
            <Paper key={q.id} elevation={0} sx={{ p: 3, border: '1px solid', borderColor: 'divider', borderRadius: 2 }}>
              <Stack spacing={2}>
                <Stack direction="row" spacing={2} alignItems="center">
                  <Typography variant="subtitle1" fontWeight="bold">
                    Q{qIndex + 1}
                  </Typography>


                  <select
                    value={q.question_type}
                    onChange={(e) => updateQuestion(qIndex, 'question_type', e.target.value)}
                    style={{ padding: '6px', borderRadius: '4px', border: '1px solid #ccc' }}
                  >
                    <option value="SINGLE">Single Choice</option>

                      style={{ width: '60px', padding: '4px', border: '1px solid #ccc', borderRadius: '4px' }}
                      min="1"
                    />

import { Navigate } from "react-router-dom";
import { useAuth } from "../hooks/useAuth";
import { Box, CircularProgress } from "@mui/material";


/**    rewrite 5-01
 *  based on auth state display route
 * @param {{ children: JSX.Element, require: 'auth' | 'guest' | 'mfa' }} props
 */
function PtcRoute({ children, require = 'auth',roles }) {
  const { authState, user } = useAuth();
    if (authState === 'LOADING') {
      return (
        <Box sx={{ minHeight: '100vh', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
          <CircularProgress />
        </Box>
      );
    }

    if (require === 'auth' && authState !== 'AUTHENTICATED') {
      if (authState === 'MFA_PENDING') return <Navigate to="/login/2fa" replace />;
      return <Navigate to="/login" replace />;
    }

    if (require === 'guest' && authState === 'AUTHENTICATED') return <Navigate to="/" replace />;
    if (require === 'guest' && authState === 'MFA_PENDING') return <Navigate to="/login/2fa" replace />;

    if (require === 'mfa' && authState !== 'MFA_PENDING') {
      return <Navigate to={authState === 'AUTHENTICATED' ? '/' : '/login'} replace />;
    }

    /*  08-01  */
    //Verify if user role is permitted for this route
    if (roles && roles.length > 0) {
      const userRole = user?.role;
      if (!userRole || !roles.includes(userRole)) {
        return <Navigate to="/" replace />; // Redirect unauthorized users to homepage
      }
    }

 
  return children; // longged in 
}

/**
 * Get nav groups visible to a role set
 * @param {string[]} userRoles
 * @returns {NavGroup[]}
 */
export function getVisibleNavGroups(userRoles) {
  return navGroups.filter((g) => g.roles.some((r) => userRoles.includes(r)));
}

/**
 * Get routes visible to a role set (for a given nav group)
 * @param {string[]} userRoles
 * @param {string} navKey
 * @returns {AppRoute[]}
 */
export function getRoutesByNav(userRoles, navKey) {
  return routesConfig.filter((r) => {
    if(r.meta.navKey !== navKey) return false;
    if(r.meta.public) return true;
    if (!r.meta.roles) return false;
    return r.meta.roles.some((role)=>userRoles.includes(role));
  });
}

/**
 * Feature flag: require special character in password
 * @type {boolean}
 */
const PASSWORD_REQUIRE_SPECIAL = false; // if require special character

//https://blog.logrocket.com/react-hook-form-vs-react-19/
/**
 * @typedef {Object} PasswordRule
 * @property {string} key
 * @property {string} label     -Hint show to user
 * @property {(v: string) => boolean} test   - Check function
 * @property {boolean} enabled  - If this rule is active
 */

/**
 * All password rules
 * @type {PasswordRule[]}
 */

export const PASSWORD_RULES =[
    { key: 'length',  label: 'At least 8 characters',        test: (v) => v.length >= 8,          enabled: true },
    { key: 'upper',   label: 'Contains an uppercase letter', test: (v) => /[A-Z]/.test(v),        enabled: true },
    { key: 'lower',   label: 'Contains a lowercase letter',  test: (v) => /[a-z]/.test(v),        enabled: true },
    { key: 'digit',   label: 'Contains a number',            test: (v) => /\d/.test(v),           enabled: true },
    { key: 'special', label: 'Contains a special character', test: (v) => /[^A-Za-z0-9]/.test(v), enabled: PASSWORD_REQUIRE_SPECIAL },
];

/**
 * get only active rules
 * @returns 
 */

export const getActiveRules = ( ) => PASSWORD_RULES.filter((rr)=>rr.enabled);


/**
 * The failed password rules 
 * @param {string} value 
 * @returns {PasswordRule[]}
 */

export const getFailedPswRules= (value) => PASSWORD_RULES.filter((rr) => rr.enabled && !rr.test(value));

/**
 * 
 * @param {string} value    - Input Password
 * @returns {string | null} -Error msg & null
 */

export const validatePassword = (value) => { 
    if(!value) return 'Please enter password';

    const failed = getFailedPswRules(value);

    if(failed.length === 0) return null;

    return failed.map((rr) => rr.label).join(' /');
 }

export const setRefresh = (v)=> localStorage.setItem(STORAGE_KEYS.refresh, v);
export const removeRefresh = ( ) => localStorage.removeItem(STORAGE_KEYS.refresh);

// User info 
export const getUser = () => {
const raw = localStorage.getItem(STORAGE_KEYS.user);
    if (!raw || raw === 'undefined' || raw === 'null') return null;
    try {return JSON.parse(raw);} catch {return null;}
};
export const setUser = (u) => {
if (!u) return;
    localStorage.setItem(STORAGE_KEYS.user, JSON.stringify(u));
};
export const removeUser = () => localStorage.removeItem(STORAGE_KEYS.user);

export const removeChallengeId = () => localStorage.removeItem(STORAGE_KEYS.challengeId);

    removeAccess();
    removeRefresh();
    removeUser();
    removeChallengeId();
};
